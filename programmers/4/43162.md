[https://school.programmers.co.kr/learn/courses/30/lessons/43162](https://school.programmers.co.kr/learn/courses/30/lessons/43162)

### 💻 코드

```
#include <string>
#include <vector>

using namespace std;

void dfs(int x, int n1, vector<bool> &visited, vector<vector<int>> &c){
    visited[x] = true;
    
    for(int i = 0; i < n1; i++){
        if(visited[i] == false && c[x][i] == 1 && x != i){
            dfs(i, n1, visited, c);
        }
    }
}

int solution(int n, vector<vector<int>> computers) {
    int answer = 0;
    
    vector<bool> visited(n, false);
    
    for(int i = 0; i < n; i++){
        if(visited[i] == false){
            dfs(i, n, visited, computers);
            answer++;
        }
    }
    return answer;
}
```
