## 백준 14502번 연구소 (C++)

이번에 풀어본 문제는 **백준 14502번 연구소**이다.

문제는 아래 링크에서 확인할 수 있다.  
[https://www.acmicpc.net/problem/14502](https://www.acmicpc.net/problem/14502)

### 💻 코드

```
#include <iostream>
#include <algorithm>

using namespace std;

int mxcnt = 0;

bool check(int x, int y, int X, int Y){
    return x >= 0 && x < X && y >= 0 && y < Y;
}

void dfs(int x, int y, int X, int Y, int (&graph)[8][8], bool (&visited)[8][8]){
    graph[x][y] = 2;
    visited[x][y] = true;

    int dx[4] = {1,-1,0,0};
    int dy[4] = {0,0,-1,1};

    for(int i = 0; i < 4; i++){
        int nx = x + dx[i];
        int ny = y + dy[i];

        if(check(nx,ny,X,Y) && !visited[nx][ny] && graph[nx][ny] == 0){
            dfs(nx,ny,X,Y,graph,visited);
        }
    }
}

void cal(int x, int y, int (&graph)[8][8]){
    bool visited[8][8] = {false};
    int g[8][8];

    int cnt = 0;

    for(int i = 0; i < x; i++){
        for(int t = 0; t < y; t++){
            g[i][t] = graph[i][t];
        }
    }

    for(int i = 0; i < x; i++){
        for(int t = 0; t < y; t++){
           if(!visited[i][t] && g[i][t] == 2){
            dfs(i,t,x,y,g,visited);
           }
        }
    }

    for(int i = 0; i < x; i++){
        for(int t = 0; t < y; t++){
           if(g[i][t] == 0){
            cnt++;
           }
        }
    }

    mxcnt = max(mxcnt,cnt);
}

void back(int X, int Y, int (&graph)[8][8], int x, int y, int cnt){
    if(cnt == 3){
        cal(X,Y,graph);
        return;
    }

    for(int i = x; i < X; i++){
        if(i == x) {
            y = y;
        }
        else{
            y = 0;
        }

        for(int t = y; t < Y; t++){
            if(graph[i][t] == 0){
                graph[i][t] = 1;
                back(X,Y,graph,i,t,cnt+1);
                graph[i][t] = 0;
            }
        }
    }
}

int main(){
    int x,y;

    mxcnt = 0;

    cin >> x >> y;

    int graph[8][8];

    for(int i = 0; i < x; i++){
        for(int t = 0; t < y; t++){
            cin >> graph[i][t];
        }
    }

    back(x,y,graph,0,0,0);

    cout << mxcnt;
}
```

### 🔍 로직 설명

일반적인 dfs문제에 백트랙킹을 중복으로 실행시켜서 구하는 문제다.

음 dfs문제인데? dfs에서 백트랙킹 하는게 아니라? 백트랙킹을 하고 그때마다 dfs를 실행시키는 더 복잡해진 문제이다

즉 벽 3개를 세우는 모든 경우의 수를 따지고 그 경우의 수마다 dfs를 돌려서 결과를 얻는 문제인데, 결국 이 코드는 시간복잡도가 (n\*m)^4이 된다 

(조합으로 3개 뽑는 복잡도 (n\*m)^3에 dfs 실행시 n\*m만큼 탐색하기 때문)

복잡도가 엄청 크다보니 이런문제는 입력값의 제한이 아주 작은것같다

백트랙킹을 하고 dfs를 한다고 생각하면 생각보다 복잡하게 구현되는 코드는 없다

back이라는 함수에서 백트랙킹 방식으로 조합으로 경우의수를 구하고(이때 순열로 구했었는데 통과는 했지만 조합으로 구하는게 더 적절하다)

최대 값을 구해야하니 cal이라는 함수에서 각각의 경우 dfs 돌려서 최신화 시키고, dfs는 기본적인 dfs처럼 짜면 된다  
(bfs가 더 적절하다고는 하는데 dfs가 더 편해서 이렇게 짰다)

백준 이지만 프로그래머스 처럼 풀기위해 최대한 전역변수를 제한했고, 전역변수를 사용하면 꼭 초기화를 시켰다.   
전역을 안쓰다보니 파라미터안에 참조형태로 값을 넘기는게 많아져서 좀 복잡해지긴했지만, 뭐 어쩔수 없다

### ✍️ 느낀 점

이 문제는 어제 소마 코테2차에서 푼 문제랑 거의 비슷하다. 근데 난 못풀었다. 왜냐? 백트랙킹을 따로 돌려서 풀생각을 안했다 

시간복잡도가 너무 높을거라 생각하시도 했고, 이 방식을 떠올리는것 자체도 힘들었다.

이 정도 수준의 문제를 많이풀어야 실력이 향상 될 것같다.
